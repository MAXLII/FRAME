cmake_minimum_required(VERSION 3.30)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" FRAME_VERSION LIMIT_COUNT 1)
project(FRAME VERSION ${FRAME_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
set(LLVM_SOURCE_DIR "${CMAKE_SOURCE_DIR}/build/deps/llvm-project/llvm" CACHE PATH "Pinned LLVM source")
set(LLVM_TARGETS_TO_BUILD "" CACHE STRING "" FORCE)
set(LLVM_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_DOCS OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_TOOLS OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_ZLIB OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_ZSTD OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_LIBXML2 OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_TERMINFO OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_PROJECTS "" CACHE STRING "" FORCE)
add_subdirectory("${LLVM_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/llvm" EXCLUDE_FROM_ALL)
add_library(frame_backend SHARED
    backend/src/api.cpp backend/src/protocol.cpp backend/src/transport.cpp
    backend/src/runtime.cpp backend/src/services.cpp backend/src/jlink.cpp backend/src/commander.cpp backend/src/symbols.cpp backend/src/tcp_transport.cpp backend/src/discovery.cpp)
configure_file(backend/src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/frame_version.rc" @ONLY)
target_sources(frame_backend PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/frame_version.rc")
target_include_directories(frame_backend PUBLIC backend/include PRIVATE backend/src)
target_include_directories(frame_backend SYSTEM PRIVATE "${LLVM_SOURCE_DIR}/include" "${CMAKE_BINARY_DIR}/llvm/include")
target_compile_definitions(frame_backend PRIVATE FRAME_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(frame_backend PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(frame_backend PRIVATE nlohmann_json::nlohmann_json LLVMObject LLVMDebugInfoDWARF LLVMSupport advapi32 ws2_32 setupapi iphlpapi)
enable_testing()
add_executable(frame_native_tests tests/native/protocol_tests.cpp backend/src/protocol.cpp)
target_include_directories(frame_native_tests PRIVATE backend/src)
target_link_libraries(frame_native_tests PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(frame_native_tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME protocol COMMAND frame_native_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/protocol-v1.json")
add_executable(frame_runtime_tests tests/native/runtime_tests.cpp backend/src/protocol.cpp)
target_include_directories(frame_runtime_tests PRIVATE backend/src)
target_link_libraries(frame_runtime_tests PRIVATE frame_backend nlohmann_json::nlohmann_json)
target_compile_options(frame_runtime_tests PRIVATE /W4 /utf-8 /EHsc)
add_executable(frame_fake_commander tests/native/fake_commander.cpp)
add_executable(frame_console_tests tests/native/console_cancel_tests.cpp)
target_compile_definitions(frame_console_tests PRIVATE NOMINMAX)
target_link_libraries(frame_console_tests PRIVATE nlohmann_json::nlohmann_json)
add_executable(frame_wire_inspect tests/native/wire_inspect.cpp backend/src/protocol.cpp)
target_include_directories(frame_wire_inspect PRIVATE backend/src)
target_link_libraries(frame_wire_inspect PRIVATE nlohmann_json::nlohmann_json)
add_dependencies(frame_runtime_tests frame_fake_commander)
add_test(NAME runtime COMMAND frame_runtime_tests "${CMAKE_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE:frame_fake_commander>")
